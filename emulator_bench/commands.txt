# CatPred EMULaToR bench commands. Run from the CatPred submodule root.
PY=/home/adhil/.conda/envs/mldb/bin/python
cd "$(git rev-parse --show-toplevel)/baselines/CatPred"
export TORCH_FORCE_NO_WEIGHTS_ONLY_LOAD=1

# 0) One-time fair-esm install, isolated from the ESM3 package.
$PY -m pip install fair-esm -t ../../outputs/baselines/CatPred/embeddings/fair_esm

# 1) Dry run: stages nothing and lists the jobs.
$PY emulator_bench/launch_parallel_retrain.py --dry_run

# 2) Full run: seeds 666 777 888, all splits, kcat km ki, original settings.
#    Stages the splits, builds the caches, trains, and resumes finished runs.
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 1

#    Multi-GPU: GPUs 0 1, two runs per GPU, bf16 mixed precision.
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 1 --runs_per_gpu 2 --amp

# 3) One task and one split, one seed.
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 --value_types kcat \
  --split_groups random_splits_grouped_sequence --seeds 666

# 4) Single steps.
$PY emulator_bench/align_structures.py --value_types kcat km ki
$PY emulator_bench/cache_embeddings.py --value_types kcat km ki

# 5) Aggregate (mean and std over seeds).
$PY emulator_bench/aggregate_catpred_results.py --seeds 666 777 888
